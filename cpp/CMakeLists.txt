cmake_minimum_required(VERSION 3.16.3)

project(sp_vision)

set(REPO_ROOT "${PROJECT_SOURCE_DIR}/..")
set(PYTHON_VENV_DIR "${REPO_ROOT}/python/.venv")
if(NOT EXISTS "${PYTHON_VENV_DIR}/bin/python" AND EXISTS "${REPO_ROOT}/.venv/bin/python")
    set(PYTHON_VENV_DIR "${REPO_ROOT}/.venv")
endif()
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release)
message(STATUS "--------------------CMAKE_BUILD_TYPE: ${CMAKE_BUILD_TYPE}--------------------")
option(BUILD_PYTHON_BINDINGS "Build pybind11 Python bindings" ON)

if(NOT fmt_DIR AND EXISTS "/usr/lib/x86_64-linux-gnu/cmake/fmt/fmt-config.cmake")
    set(fmt_DIR "/usr/lib/x86_64-linux-gnu/cmake/fmt")
endif()

if(NOT yaml-cpp_DIR AND EXISTS "/usr/lib/x86_64-linux-gnu/cmake/yaml-cpp/yaml-cpp-config.cmake")
    set(yaml-cpp_DIR "/usr/lib/x86_64-linux-gnu/cmake/yaml-cpp")
endif()

if(NOT nlohmann_json_DIR AND EXISTS "/usr/lib/cmake/nlohmann_json/nlohmann_jsonConfig.cmake")
    set(nlohmann_json_DIR "/usr/lib/cmake/nlohmann_json")
endif()

find_package(OpenCV REQUIRED)
find_package(fmt REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(spdlog REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(nlohmann_json REQUIRED)
if(BUILD_PYTHON_BINDINGS)
    if(EXISTS "${PYTHON_VENV_DIR}/bin/python" AND NOT Python3_EXECUTABLE)
        set(Python3_EXECUTABLE "${PYTHON_VENV_DIR}/bin/python")
    endif()
    find_package(Python3 COMPONENTS Interpreter Development QUIET)
    if(NOT EXISTS "${PYTHON_VENV_DIR}/bin/python")
        message(WARNING "Project .venv not found, Python bindings will be skipped.")
        set(BUILD_PYTHON_BINDINGS OFF)
    elseif(NOT Python3_FOUND)
        message(WARNING "Python development headers not found, Python bindings will be skipped.")
        set(BUILD_PYTHON_BINDINGS OFF)
    endif()
endif()
if(NOT OpenVINO_DIR)
    file(GLOB _openvino_runtime_dirs "/opt/intel/openvino_*/runtime/cmake")
    if(_openvino_runtime_dirs)
        list(SORT _openvino_runtime_dirs)
        list(LENGTH _openvino_runtime_dirs _openvino_runtime_count)
        math(EXPR _openvino_runtime_index "${_openvino_runtime_count} - 1")
        list(GET _openvino_runtime_dirs ${_openvino_runtime_index} OpenVINO_DIR)
    else()
        file(GLOB _openvino_cmake_dirs "/usr/lib/cmake/openvino*")
        if(_openvino_cmake_dirs)
            list(SORT _openvino_cmake_dirs)
            list(LENGTH _openvino_cmake_dirs _openvino_cmake_count)
            math(EXPR _openvino_cmake_index "${_openvino_cmake_count} - 1")
            list(GET _openvino_cmake_dirs ${_openvino_cmake_index} OpenVINO_DIR)
        endif()
    endif()
endif()
if(OpenVINO_DIR)
    message(STATUS "Using OpenVINO_DIR=${OpenVINO_DIR}")
endif()
find_package(OpenVINO REQUIRED)

include_directories(${EIGEN3_INCLUDE_DIR})
include_directories(${OpenCV_INCLUDE_DIRS})
include_directories(${PROJECT_SOURCE_DIR})

add_subdirectory(tools)
add_subdirectory(io)
add_subdirectory(tasks/auto_aim)
add_subdirectory(tasks/auto_buff)
add_subdirectory(tasks/omniperception)
add_subdirectory(apps)
add_subdirectory(tests)
if(BUILD_PYTHON_BINDINGS)
    add_subdirectory(bindings)
endif()
